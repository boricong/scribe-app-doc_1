\begin{abstract}
This document is a compact test suite for a \LaTeX{} editor.
It contains sectioning commands, mathematical notation, theorem
environments, cross-references, lists, tables, internally generated
graphics, source-code listings, hyperlinks, citations, footnotes,
and a bibliography. No external image, font, bibliography, or data
file is required. Explicit page breaks are used so that the document
has a predictable three-page structure.
\end{abstract}

\subsection{Definitions and Theorems}

\begin{definition}[Convex function]
A function $begin:math:text$f\:\\R\^d\\rightarrow\\R$end:math:text$ is convex when, for every
$begin:math:text$\\vect\{x\}\,\\vect\{y\}\\in\\R\^d$end:math:text$ and $begin:math:text$t\\in\[0\,1\]$end:math:text$,
$begin:math:display$
  f\\bigl\(t\\vect\{x\}\+\(1\-t\)\\vect\{y\}\\bigr\)
  \\leq
  t f\(\\vect\{x\}\)\+\(1\-t\)f\(\\vect\{y\}\)\.
$end:math:display$
\end{definition}

\begin{lemma}
\label{lem:square}
For every real number $begin:math:text$x$end:math:text$, one has $begin:math:text$x\^2\\geq 0$end:math:text$.
\end{lemma}

\begin{proof}
If $begin:math:text$x\\geq0$end:math:text$, then $begin:math:text$x\^2\=x\\cdot x\\geq0$end:math:text$.
If $begin:math:text$x\<0$end:math:text$, then $begin:math:text$\-x\>0$end:math:text$, and
$begin:math:text$x\^2\=\(\-x\)\^2\\geq0$end:math:text$.
\end{proof}

\begin{theorem}[Mean minimization]
\label{thm:mean}
Let $begin:math:text$x\_1\,\\ldots\,x\_n\\in\\R$end:math:text$. The function
$begin:math:display$
  Q\(c\)\=\\sum\_\{i\=1\}\^\{n\}\(x\_i\-c\)\^2
$end:math:display$
is minimized at the arithmetic mean
$begin:math:text$\\bar\{x\}\=n\^\{\-1\}\\sum\_\{i\=1\}\^\{n\}x\_i$end:math:text$.
\end{theorem}

\begin{proof}
Write $begin:math:text$x\_i\-c\=\(x\_i\-\\bar\{x\}\)\+\(\\bar\{x\}\-c\)$end:math:text$. Expanding gives
\begin{align*}
  Q(c)
    &=
    \sum_{i=1}^{n}(x_i-\bar{x})^2\\
    &\quad
    +2(\bar{x}-c)\sum_{i=1}^{n}(x_i-\bar{x})\\
    &\quad
    +n(\bar{x}-c)^2.
\end{align*}
The middle term is zero by the definition of $begin:math:text$\\bar\{x\}$end:math:text$.
The last term is nonnegative by Lemma~\ref{lem:square} and is zero
exactly when $begin:math:text$c\=\\bar\{x\}$end:math:text$.
\end{proof}

\end{remark}

\subsection{Algorithmic Pseudocode}

The following source-code listing tests indentation, line numbering,
comments, mathematical-looking identifiers, and line wrapping.

\begin{lstlisting}[language=Python,caption={A compact gradient-descent example.},label={lst:gradient}]
def gradient_descent(x0, learning_rate=0.1,
                     tolerance=1e-8, max_iter=1000):
    """Minimize f(x) = (x - 3)^2."""
    x = float(x0)

    for iteration in range(max_iter):
        gradient = 2.0 * (x - 3.0)
        candidate = x - learning_rate * gradient

        # Stop when the update is sufficiently small.
        if abs(candidate - x) < tolerance:
            return candidate, iteration + 1

        x = candidate

    return x, max_iter
\end{lstlisting}

Listing~\ref{lst:gradient} is not executed by \LaTeX; it is included
only to test syntax highlighting and verbatim-like content.
Characters that normally have special meanings, such as
\verb|#|, \verb|_|, \verb|%|, and \verb|\|, can be printed inside
the \verb|\verb| command.
